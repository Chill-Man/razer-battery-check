cmake_minimum_required(VERSION 3.16)
project(BasiliskPulse VERSION 1.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  message(FATAL_ERROR "Use build.ps1 for Windows releases.")
endif()
find_package(Threads REQUIRED)
find_package(Qt5 5.12 REQUIRED COMPONENTS Widgets Network)
set(CMAKE_AUTORCC ON)
add_library(pulse_core src/battery_core.cpp src/battery_linux.cpp)
target_include_directories(pulse_core PUBLIC src)
target_link_libraries(pulse_core PUBLIC Threads::Threads)
target_compile_options(pulse_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(basilisk-pulse src/linux_main.cpp src/linux_resources.qrc)
target_link_libraries(basilisk-pulse PRIVATE pulse_core Qt5::Widgets Qt5::Network)
target_compile_definitions(basilisk-pulse PRIVATE PULSE_VERSION="${PROJECT_VERSION}")
add_executable(pulse-tests tests/tests.cpp)
target_link_libraries(pulse-tests PRIVATE pulse_core)
enable_testing()
add_test(NAME battery-regression COMMAND pulse-tests)
install(TARGETS basilisk-pulse RUNTIME DESTINATION bin)
install(FILES packaging/linux/basilisk-pulse.desktop DESTINATION share/applications)
install(FILES packaging/linux/basilisk-pulse.png DESTINATION share/icons/hicolor/256x256/apps)
install(FILES LICENSE DESTINATION share/doc/basilisk-pulse)
